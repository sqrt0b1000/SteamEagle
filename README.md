# Steam Eagle

`SwiftUI` like `HTML` rendere,
that focuses on speeeeeed.

## Benchmark

```zsh
swift package benchmark --target BenchmarkHTMLKit
```

```swift
// Random data String
let data = (0...10000).map { _ in
    String.init(repeating: Int.random(in: 0...10_000_000).description, count: 10)
}

// Structure of the view (header with content)
struct TestHTMLKit: View {
    let x: String

    @HTMLKit.ContentBuilder<HTMLKit.Content>
    var body: any HTMLKit.Content {
        Heading1 {
            String(repeating: x, count: 50)
        }
    }
}

let htmlkitRenderer = HTMLKit.Renderer()

// in the benchmark

for x in 0...10000 {
    try htmlkitRenderer.render(view: TestHTMLKit(x: data[x]))
}

```
